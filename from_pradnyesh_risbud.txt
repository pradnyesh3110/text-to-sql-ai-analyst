<script>
  function showPage(name) {
    document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
    document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
    document.getElementById('page-' + name).classList.add('active');
    document.getElementById('nav-' + name).classList.add('active');
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }

  function togglePoem(card) {
    const body = card.querySelector('.poem-body');
    const isOpen = body.classList.contains('open');
    document.querySelectorAll('.poem-body').forEach(b => b.classList.remove('open'));
    if (!isOpen) {
      body.classList.add('open');
      spawnHearts(card);
    }
  }

  function spawnHearts(element) {
    const rect = element.getBoundingClientRect();
    const hearts = ['💛', '🧸', '✨', '🌸', '💌'];
    for (let i = 0; i < 5; i++) {
      setTimeout(() => {
        const heart = document.createElement('div');
        heart.className = 'heart-float';
        heart.textContent = hearts[Math.floor(Math.random() * hearts.length)];
        heart.style.left = (rect.left + Math.random() * rect.width) + 'px';
        heart.style.top = (rect.top + rect.height / 2) + 'px';
        heart.style.fontSize = (16 + Math.random() * 12) + 'px';
        document.body.appendChild(heart);
        setTimeout(() => heart.remove(), 4000);
      }, i * 150);
    }
  }

  // Auto-play gentle heart animation on home page
  setInterval(() => {
    if (document.getElementById('page-home').classList.contains('active')) {
      const heart = document.createElement('div');
      heart.className = 'heart-float';
      heart.textContent = ['💛', '🧸', '✨'][Math.floor(Math.random() * 3)];
      heart.style.left = (Math.random() * window.innerWidth) + 'px';
      heart.style.top = (window.innerHeight - 20) + 'px';
      heart.style.fontSize = (14 + Math.random() * 10) + 'px';
      document.body.appendChild(heart);
      setTimeout(() => heart.remove(), 4000);
    }
  }, 3000);
</script>
